let syncTimer = null; window.onload = function() { document.getElementById('datePicker').value = selectedDate; document.getElementById('datePicker').addEventListener('change', (e) => { selectedDate = e.target.value; updateCurrentReservationMap(); renderGrid(); }); renderZones(); loadData(); // 开启定时同步（15 秒一次） startSync(); // 监测手机状态：切后台/切微信/锁屏时自动暂停请求，回到页面时立即刷新并恢复 document.addEventListener("visibilitychange", function() { if (document.hidden) { stopSync(); } else { loadData(); // 切回前台第一时间刷新 startSync(); } }); }; function startSync() { if (!syncTimer) { syncTimer = setInterval(loadData, 15000); // 15秒/次 } } function stopSync() { if (syncTimer) { clearInterval(syncTimer); syncTimer = null; } }
